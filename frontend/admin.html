async function loadStats() { try { const token = localStorage.getItem('adminToken'); const r = await fetch(API_URL + '/api/admin/stats', { headers: { 'Authorization': 'Bearer ' + token } }); const d = await r.json(); document.getElementById('totalUsers').textContent = d.totalUsers || 0; document.getElementById('totalDeployments').textContent = d.totalDeployments || 0; document.getElementById('activeDeployments').textContent = d.activeDeployments || 0; document.getElementById('totalVisits').textContent = d.totalVisits || 0; } catch (e) { console.error(e); } } async function loadUsers() { try { const token = localStorage.getItem('adminToken'); const r = await fetch(API_URL + '/api/admin/users', { headers: { 'Authorization': 'Bearer ' + token } }); const d = await r.json(); console.log('Users from API:', d); allUsers = d.users || []; renderUsers(allUsers); loadStats(); } catch (e) { console.error('Load users error:', e); showToast('❌ Error loading users: ' + e.message, 'error'); } }
